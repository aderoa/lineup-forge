NBA Stats requires ALL parameters present (even empty), and uses raw API names (MeasureType, PerMode, etc.) not the Python-style names that nba_api's class wrapper exposes.""" return { 'GroupQuantity': 5, 'LastNGames': 0, 'LeagueID': '00', 'MeasureType': 'Advanced', 'Month': 0, 'OpponentTeamID': 0, 'PaceAdjust': 'N', 'PerMode': 'Per100Possessions', 'Period': 0, 'PlusMinus': 'N', 'Rank': 'N', 'Season': season, 'SeasonType': season_type, 'PlayerID': player_id, # <-- the filter that nba_api's wrapper doesn't expose 'TeamID': 0, 'GameSegment': '', 'DateFrom': '', 'DateTo': '', 'GameID': '', 'Outcome': '', 'SeasonSegment': '', 'VsConference': '', 'VsDivision': '', 'Location': '', 'ShotClockRange': '', 'PORound': 0, 'Conference': '', 'Division': '', } def fetch_player_lineups(player_id, player_name, season, season_type, max_retries=4, retry_delay=10): """Fetch up to 250 lineups featuring a specific player for (season, season_type). Uses NBA Stats' leaguedashlineups endpoint with PlayerID filter, called via nba_api's NBAStatsHTTP transport (so we get the right headers + retry behavior without depending on the LeagueDashLineups class wrapper which doesn't expose player_id).""" safe_name = ''.join(c for c in player_name if c.isalnum() or c in '_-').strip() cache_file = (CACHE_DIR / f"{player_id}__{safe_name}__{season}__{season_type.replace(' ', '-')}.csv") if cache_file.exists() and cache_file.stat().st_size > 100: return pd.read_csv(cache_file), True parameters = _lineup_api_params(player_id, season, season_type) for attempt in range(max_retries): try: response = NBAStatsHTTP().send_api_request( endpoint='leaguedashlineups', parameters=parameters, timeout=90, ) data = response.get_dict() result_sets = data.get('resultSets', []) if not result_sets: empty = pd.DataFrame() empty.to_csv(cache_file, index=False) return empty, False # Annotate provenance so dedupe can drop multi-player duplicates later df.insert(0, 'SOURCE_PLAYER_NAME', player_name) df.insert(0, 'SOURCE_PLAYER_ID', player_id) df.insert(0, 'SEASON_TYPE', season_type) df.insert(0, 'SEASON', season) df.to_csv(cache_file, index=False) return df, False except (ReqConnectionError, ReadTimeout, Timeout) as e: wait = (attempt + 1) * retry_delay print(f" ⚠ {type(e).__name__} — retry {attempt+1}/{max_retries} in {wait}s") time.sleep(wait) except JSONDecodeError: wait = (attempt + 1) * retry_delay * 2 print(f" ⚠ JSON error (throttle) — retry {attempt+1}/{max_retries} in {wait}s") time.sleep(wait) except Exception as e: wait = (attempt + 1) * retry_delay print(f" ⚠ {type(e).__name__}: {str(e)[:80]} — retry {attempt+1}/{max_retries} in {wait}s") time.sleep(wait) print(f" ✗ Terminal failure after {max_retries} retries") return None, False def fetch_all(args): """Iterate over (player × season × season_type) and cache each slice.""" post_dedupe:,} duplicates removed)") # Drop the source-player provenance columns — they're per-fetch metadata, not lineup attributes drop_cols = [c for c in ('SOURCE_PLAYER_ID', 'SOURCE_PLAYER_NAME') if c in merged.columns] if drop_cols: merged = merged.drop(columns=drop_cols) # Reorder: meta columns first meta_order = ['TEAM_ABBREVIATION', 'SEASON', 'SEASON_TYPE'] meta_present = [c for c in meta_order if c in merged.columns] rest = [c for c in merged.columns if c not in meta_present] merged = merged[meta_present + rest] print(f" 📊 lineups_5man {len(merged):>7,} rows × {len(merged.columns):>3} cols") return merged def write_sqlite(df, path, gzip_output=True): """Write the consolidated lineups into SQLite with indexes. Same schema as nba_lineups_bulk.py so the app needs no changes.""" p = Path(path) if p.exists(): p.unlink() INDEX_SINGLE = ['TEAM_ABBREVIATION', 'SEASON', 'SEASON_TYPE', 'GROUP_ID', 'GROUP_NAME'] df.fillna('').replace([float('inf'), float('-inf')], '') df_clean.to_sql('lineups_5man', conn, if_exists='replace', index=False) print(f" • lineups_5man {len(df_clean):,} rows × {len(df_clean.columns)} cols") cursor = conn.cursor() n_idx = 0 for col in INDEX_SINGLE: if col in df_clean.columns: cursor.execute(f'CREATE INDEX "idx_lineups_5man_{col.lower()}" ON "lineups_5man"("{col}")') n_idx += 1 for cols in INDEX_COMPOUND: if all(c in df_clean.columns for c in cols): idx_name = f"idx_lineups_5man_{'_'.join(c.lower() for c in cols)}" cols_sql = ', '.join(f'"{c}"' for c in cols) cursor.execute(f'CREATE INDEX "{idx_name}" ON "lineups_5man"({cols_sql})') n_idx += 1 conn.commit() print(f" • {n_idx} indexes built") meta_df = pd.DataFrame([ ('built_at', datetime.now().strftime('%Y-%m-%d %H:%M:%S')), ('script', 'nba_player_lineups.py'), ('extraction_mode', 'player-centric'), ('volume_threshold_min', str(MIN_VOLUME_THRESHOLD)), ('current_season', CURRENT_SEASON), ('total_rows', str(len(df_clean))), ], columns=['key', 'value']) meta_df.to_sql('_meta', conn, if_exists='replace', index=False) print(" • running VACUUM...", end=' ', flush=True) conn.execute("VACUUM") conn.close() print('✓') raw_mb = p.stat().st_size / (1024 * 1024) print(f"\n✅ {path} — {len(df_clean):,} rows, {raw_mb:.1f} MB on disk") if gzip_output: gz_path = str(p) + '.gz' with open(p, 'rb') as f_in, gzip.open(gz_path, 'wb', compresslevel=9) as f_out: shutil.copyfileobj(f_in, f_out) gz_mb = Path(gz_path).stat().st_size / (1024 * 1024) print(f"✅ {gz_path} — {gz_mb:.1f} MB gzipped ({raw_mb/gz_mb:.1f}x compression)") return True def main(): p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter) p.add_argument('--start', type=int, default=DEFAULT_START_SEASON, help=f'Start season year (default: {DEFAULT_START_SEASON} = {season_str(DEFAULT_START_SEASON)})') p.add_argument('--end', type=int,
